// Step 2 · Show current database
db

db.createCollection("students")

// Step 3 · Show collections
show collections

// Step 4 · insertOne
db.students.insertOne({ rollNo: 101, name: "Arun", department: "CSE", age: 20, mark: 85 })

// Step 4 · insertMany
db.students.insertMany([
  { rollNo: 102, name: "Priya", department: "IT", age: 21, mark: 92 },
  { rollNo: 103, name: "Kumar", department: "CSE", age: 19, mark: 78 },
  { rollNo: 104, name: "Divya", department: "ECE", age: 20, mark: 88 },
  { rollNo: 105, name: "Ravi", department: "CSE", age: 22, mark: 67 }
])

// Step 5 · Display all records
db.students.find().pretty()

// Step 5 · Find by name
db.students.find({ name: "Priya" })

// Step 5 · findOne from CSE
db.students.findOne({ department: "CSE" })

// Step 6 · Marks greater than 80
db.students.find({ mark: { $gt: 80 } })

// Step 6 · Age between 20 and 21
db.students.find({ age: { $gte: 20, $lte: 21 } })

// Step 6 · CSE or IT
db.students.find({ department: { $in: ["CSE", "IT"] } })

// Step 6 · Marks less than 80
db.students.find({ mark: { $lt: 80 } })

// Step 7 · Update Arun's marks
db.students.updateOne({ rollNo: 101 }, { $set: { mark: 90 } })

// Step 7 · +5 marks for all CSE
db.students.updateMany({ department: "CSE" }, { $inc: { mark: 5 } })

// Step 7 · Verify updates
db.students.find()

// Step 8 · deleteOne (roll 105)
db.students.deleteOne({ rollNo: 105 })

// Step 8 · deleteMany (marks below 70)
db.students.deleteMany({ mark: { $lt: 70 } })

// Step 9 · Sort by marks (high to low)
db.students.find().sort({ mark: -1 })

// Step 9 · Top three students
db.students.find().sort({ mark: -1 }).limit(3)

// Step 10 · Count all students
db.students.countDocuments()

// Step 10 · Count CSE students
db.students.countDocuments({ department: "CSE" })

// Step 11 · Show databases
show dbs

// Step 11 · Database statistics
db.stats()
